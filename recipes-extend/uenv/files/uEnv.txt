        mmc_overlays=dtb/renesas/overlays
        mmc_args=setenv bootargs 'rw rootwait earlycon root=/dev/mmcblk0p2'
        prodsdboot=run mmc_args; run mmc_image; run mmc_ovfdt;booti $image_addr - $dtb_addr;
        mmc_image=fatload mmc 0:1 0x48080000 Image
        mmc_fdt=if test "${board_id}" = "rzg2l-sbc"; then setenv fdtfile rzg2l-sbc.dtb; elif test "${board_id}" = "rzg2l-evk"; then setenv fdtfile r9a07g044l2-smarc.dtb; elif test "${board_id}" = "rzv2l-evk"; then setenv fdtfile r9a07g054l2-smarc.dtb; elif test "${board_id}" = "rzv2h-evk"; then setenv fdtfile r9a09g057h4-evk-ver1.dtb; fi; fatload mmc 0:1 0x48000000 dtb/renesas/${fdtfile}
        mmc_ovfdt=if run mmc_fdt; then fdt addr ${dtb_addr}; fdt resize 0x10000; if env exists enable_overlay_i2c &&    test ${enable_overlay_i2c} = 1 -o ${enable_overlay_i2c} = yes ; then echo Applying DT overlay: rzg2l-sbc-ext-i2c.dtbo ;fatload mmc 0:1 ${dtbo_addr} ${mmc_overlays}/rzg2l-sbc-ext-i2c.dtbo; fdt addr ${dtb_addr}; fdt apply ${dtbo_addr}; fi; if env exists enable_overlay_spi &&    test ${enable_overlay_spi} = 1 -o ${enable_overlay_spi} = yes ; then echo Applying DT overlay: rzg2l-sbc-ext-spi.dtbo ;fatload mmc 0:1 ${dtbo_addr} ${mmc_overlays}/rzg2l-sbc-ext-spi.dtbo; fdt addr ${dtb_addr}; fdt apply ${dtbo_addr}; fi; if env exists enable_overlay_can &&    test ${enable_overlay_can} = 1 -o ${enable_overlay_can} = yes ; then echo Applying DT overlay: rzg2l-sbc-can.dtbo ;fatload mmc 0:1 ${dtbo_addr} ${mmc_overlays}/rzg2l-sbc-can.dtbo; fdt addr ${dtb_addr}; fdt apply ${dtbo_addr}; fi; if env exists enable_overlay_dsi &&    test ${enable_overlay_dsi} = 1 -o ${enable_overlay_dsi} = yes ; then echo Applying DT overlay: rzg2l-sbc-dsi.dtbo ;fatload mmc 0:1 ${dtbo_addr} ${mmc_overlays}/rzg2l-sbc-dsi.dtbo; fdt addr ${dtb_addr}; fdt apply ${dtbo_addr}; fi; if env exists enable_overlay_csi_ov5640 &&    test ${enable_overlay_csi_ov5640} = 1 -o ${enable_overlay_csi_ov5640} = yes ; then echo Applying DT overlay: rzg2l-sbc-ov5640.dtbo ;fatload mmc 0:1 ${dtbo_addr} ${mmc_overlays}/rzg2l-sbc-ov5640.dtbo; fdt addr ${dtb_addr}; fdt apply ${dtbo_addr}; fi; else echo WARN: Cannot load the DT; fi;
